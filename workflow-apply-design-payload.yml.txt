# ".github/workflows/apply-design-payload.yml" -> Commit changes.

name: Apply design payload

on:
  push:
    branches: [main]
    paths: ['design-payload.json']

permissions:
  contents: write

jobs:
  apply:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
      - uses: actions/setup-node@v6
        with:
          node-version: 24
      - name: Baixar e verificar arquivos do payload
        run: node scripts/apply-payload.mjs
      - name: Validar build e integracao
        run: npm test
      - name: Commit e push
        run: |
          git config user.name "apecerto-design-bot"
          git config user.email "design-bot@users.noreply.github.com"
          git rm -q design-payload.json
          git add -A
          git commit -m "Aplica atualizacao de design via payload"
          git push
